`select r.*, v.name as version_name,` ` (select count(1) from ai_eval_result x where x.run_id = r.id) as done,` ` (select count(1) from ai_eval_result x where x.run_id = r.id and x.outcome = 'CORRECT') as correct,` ` (select count(1) from ai_eval_result x where x.run_id = r.id and x.outcome = 'ERROR') as errors` ` from ai_eval_run r left join ai_model_version v on v.id = r.version_id`
